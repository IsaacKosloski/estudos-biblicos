\chapter{A pergunta}

\begin{objetivo}
Formular a pergunta com precisão, dividi-la em partes e registrar o que já
penso antes de começar, para poder comparar no fim.
\end{objetivo}

\begin{campo}{A pergunta do estudo}
\temaPergunta
\end{campo}

\begin{campo}{Subperguntas (proposta inicial)}
\begin{enumerate}
  \item O que o sangue representa nos textos bíblicos?
  \item Que orientações são dadas sobre o sangue, a quem e em que momento?
  \item Que papel o sangue tem nos sacrifícios e no significado da morte de Jesus?
  \item Como os textos sobre derramamento de sangue se relacionam com o valor da vida?
\end{enumerate}
\end{campo}

\campovazio{O que já penso}{e de onde vem essa ideia (leitura, ensino recebido, experiência)}

\begin{campo}{Escopo}
Toda a Bíblia, com atenção à passagem da Lei mosaica para o Novo Testamento.
\end{campo}
