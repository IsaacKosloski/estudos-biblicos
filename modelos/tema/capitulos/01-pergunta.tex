\chapter{A pergunta}

\begin{objetivo}
Formular a pergunta com precisão, dividi-la em partes e registrar o que já
penso antes de começar, para poder comparar no fim.
\end{objetivo}

\begin{campo}{A pergunta do estudo}
\temaPergunta
\end{campo}

\campovazio{Subperguntas}{em que partes a pergunta se divide?}
\campovazio{O que já penso}{e de onde vem essa ideia (leitura, ensino recebido, experiência)}
\campovazio[2]{Escopo}{toda a Bíblia? só um testamento? só um livro?}
